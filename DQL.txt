-- Select Database
USE CompanyDB;

-- 1. List the number of employees and average salary for employees in department 20.
SELECT COUNT(*) AS No_of_Employees,
       AVG(SAL) AS Average_Salary
FROM EMP
WHERE DEPTNO = 20;

-- 2. List name, salary and PF amount of all employees. (PF = 10% of Salary)
SELECT ENAME,
       SAL,
       SAL * 0.10 AS PF
FROM EMP;

-- 3. List the employee details in the ascending order of their basic salary.
SELECT *
FROM EMP
ORDER BY SAL ASC;

-- 4. List the employee name and hire date in the descending order of the hire date.
SELECT ENAME,
       HIREDATE
FROM EMP
ORDER BY HIREDATE DESC;

-- 5. List employee name, salary, PF, HRA, DA and Gross; order the results in ascending order of Gross.
SELECT ENAME,
       SAL,
       SAL * 0.10 AS PF,
       SAL * 0.50 AS HRA,
       SAL * 0.30 AS DA,
       (SAL + (SAL * 0.50) + (SAL * 0.30)) AS Gross
FROM EMP
ORDER BY Gross ASC;

-- 6. List the department numbers and number of employees in each department.
SELECT DEPTNO,
       COUNT(*) AS No_of_Employees
FROM EMP
GROUP BY DEPTNO;

-- 7. Increment the salary of Salesman by 10% of basic salary.
UPDATE EMP
SET SAL = SAL + (SAL * 0.10)
WHERE JOB = 'SALESMAN';

-- 8. List the total salary, maximum salary, minimum salary and average salary of employees for department 20.
SELECT SUM(SAL) AS Total_Salary,
       MAX(SAL) AS Maximum_Salary,
       MIN(SAL) AS Minimum_Salary,
       AVG(SAL) AS Average_Salary
FROM EMP
WHERE DEPTNO = 20;

-- 9. List the employees whose names contain 'I' as the third letter.
SELECT *
FROM EMP
WHERE ENAME LIKE '__I%';

-- 10. List the maximum salary paid to a Salesman.
SELECT MAX(SAL) AS Maximum_Salary
FROM EMP
WHERE JOB = 'SALESMAN';

-- 11. Increase the salary of Salesman by 10% of their current salary.
UPDATE EMP
SET SAL = SAL * 1.10
WHERE JOB = 'SALESMAN';
