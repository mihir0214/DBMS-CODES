
USE CompanyDB;

-- 1. INSERT A NEW EMPLOYEE
INSERT INTO EMP (EMPNO, ENAME, JOB, SAL, DEPTNO)
VALUES (1010, 'RAHUL', 'CLERK', 15000, 20);


-- 2. INSERT MULTIPLE EMPLOYEES
INSERT INTO EMP (EMPNO, ENAME, JOB, SAL, DEPTNO)
VALUES
(1011, 'AMIT', 'MANAGER', 30000, 20),
(1012, 'PRIYA', 'ANALYST', 25000, 30);


-- 3. UPDATE EMPLOYEE SALARY
UPDATE EMP
SET SAL = 25000
WHERE EMPNO = 1010;


-- 4. INCREASE SALARY BY 10%
UPDATE EMP
SET SAL = SAL * 1.10
WHERE JOB = 'SALESMAN';


-- 5. UPDATE JOB
UPDATE EMP
SET JOB = 'MANAGER'
WHERE EMPNO = 1011;


-- 6. UPDATE DEPARTMENT NUMBER
UPDATE EMP
SET DEPTNO = 30
WHERE EMPNO = 1012;


-- 7. DELETE ONE EMPLOYEE
DELETE FROM EMP
WHERE EMPNO = 1010;


-- 8. DELETE ALL SALESMAN RECORDS
DELETE FROM EMP
WHERE JOB = 'SALESMAN';


-- 9. INSERT DATA FROM ANOTHER TABLE
INSERT INTO EMP_BACKUP
SELECT * FROM EMP;


-- 10. DELETE EMPLOYEES WITH SALARY BELOW 10000
DELETE FROM EMP
WHERE SAL < 10000;


-- DISPLAY FINAL RECORDS
SELECT * FROM EMP;
