TO 'rahul'@'localhost';

TO 'rahul'@'localhost';

TO 'rahul'@'localhost';


-- 6. GRANT ALL PRIVILEGES
GRANT ALL PRIVILEGES ON CompanyDB.*
TO 'rahul'@'localhost';
