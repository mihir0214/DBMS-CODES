-- CREATE DATABASE
CREATE DATABASE IF NOT EXISTS CompanyDB;
USE CompanyDB;

-- CREATE EMPLOYEE TABLE
CREATE TABLE EMP (
    EMPNO INT PRIMARY KEY,
    ENAME VARCHAR(30),
    JOB VARCHAR(20),
    SAL DECIMAL(10,2),
    DEPTNO INT
);

DELIMITER //

CREATE FUNCTION ADD_NUM(a INT, b INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN a + b;
