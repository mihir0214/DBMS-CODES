-- 1. CREATE MAIN EMPLOYEE TABLE

CREATE TABLE EMP (
    EMPNO INT PRIMARY KEY,
    ENAME VARCHAR(50),
    SAL DECIMAL(10,2),
    DEPTNO INT
);


-- 2. CREATE LOG TABLE

CREATE TABLE EMP_LOG (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    EMPNO INT,
    ACTION_TYPE VARCHAR(30),
    OLD_SAL DECIMAL(10,2),
    NEW_SAL DECIMAL(10,2),
    ACTION_DATE DATETIME
);


DELIMITER //


-- 3. BEFORE INSERT TRIGGER

CREATE TRIGGER before_emp_insert
BEFORE INSERT ON EMP
FOR EACH ROW
BEGIN
    SET NEW.ENAME = UPPER(NEW.ENAME);
END//


-- 4. AFTER INSERT TRIGGER

CREATE TRIGGER after_emp_insert
AFTER INSERT ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_LOG
    (EMPNO, ACTION_TYPE, OLD_SAL, NEW_SAL, ACTION_DATE)
    VALUES
    (NEW.EMPNO, 'AFTER INSERT', NULL, NEW.SAL, NOW());
END//


-- 5. BEFORE UPDATE TRIGGER

CREATE TRIGGER before_emp_update
BEFORE UPDATE ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_LOG
    (EMPNO, ACTION_TYPE, OLD_SAL, NEW_SAL, ACTION_DATE)
    VALUES
    (OLD.EMPNO, 'BEFORE UPDATE', OLD.SAL, NEW.SAL, NOW());
END//


-- 6. AFTER UPDATE TRIGGER

CREATE TRIGGER after_emp_update
AFTER UPDATE ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_LOG
    (EMPNO, ACTION_TYPE, OLD_SAL, NEW_SAL, ACTION_DATE)
    VALUES
    (NEW.EMPNO, 'AFTER UPDATE', OLD.SAL, NEW.SAL, NOW());
END//


-- 7. BEFORE DELETE TRIGGER

CREATE TRIGGER before_emp_delete
BEFORE DELETE ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_LOG
    (EMPNO, ACTION_TYPE, OLD_SAL, NEW_SAL, ACTION_DATE)
    VALUES
    (OLD.EMPNO, 'BEFORE DELETE', OLD.SAL, NULL, NOW());
END//


-- 8. AFTER DELETE TRIGGER

CREATE TRIGGER after_emp_delete
AFTER DELETE ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_LOG
    (EMPNO, ACTION_TYPE, OLD_SAL, NEW_SAL, ACTION_DATE)
    VALUES
    (OLD.EMPNO, 'AFTER DELETE', OLD.SAL, NULL, NOW());
END//


DELIMITER ;


-- 9. INSERT EMPLOYEE

INSERT INTO EMP VALUES (101, 'rahul', 30000, 10);
INSERT INTO EMP VALUES (102, 'amit', 25000, 20);


-- 10. UPDATE EMPLOYEE

UPDATE EMP
SET SAL = 40000
WHERE EMPNO = 101;


-- 11. DELETE EMPLOYEE

DELETE FROM EMP
WHERE EMPNO = 102;


-- 12. DISPLAY MAIN TABLE

SELECT * FROM EMP;


-- 13. DISPLAY LOG TABLE

SELECT * FROM EMP_LOG;
