-- DISPLAY ALL EMPLOYEES USING CURSOR

DELIMITER //

CREATE PROCEDURE CURSOR_1()
BEGIN
    DECLARE DONE INT DEFAULT 0;
    DECLARE E_NAME VARCHAR(30);

    DECLARE C1 CURSOR FOR
        SELECT ENAME FROM EMP;

    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET DONE = 1;

    OPEN C1;

    L1: LOOP
        FETCH C1 INTO E_NAME;

        IF DONE = 1 THEN
            LEAVE L1;
        END IF;

        SELECT E_NAME;
    END LOOP;

    CLOSE C1;
END //

DELIMITER ;

CALL CURSOR_1();
