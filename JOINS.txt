
-- CREATE TABLES

CREATE TABLE student (
    id INT PRIMARY KEY,
    name VARCHAR(30),
    city VARCHAR(30)
);

CREATE TABLE marks (
    id INT,
    subject VARCHAR(30),
    marks INT
);

-- 1. INNER JOIN

SELECT student.id, student.name, marks.subject, marks.marks
FROM student
INNER JOIN marks
ON student.id = marks.id;


-- 2. LEFT JOIN

SELECT student.id, student.name, marks.subject, marks.marks
FROM student
LEFT JOIN marks
ON student.id = marks.id;


-- 3. RIGHT JOIN

SELECT student.id, student.name, marks.subject, marks.marks
FROM student
RIGHT JOIN marks
ON student.id = marks.id;


-- 4. CROSS JOIN

SELECT student.name, marks.subject
FROM student
CROSS JOIN marks;


-- 5. SELF JOIN

CREATE TABLE employee (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(30),
    manager_id INT
);

-- 6. NATURAL JOIN

SELECT *
FROM student
